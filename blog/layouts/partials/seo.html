{{- /* Centralised SEO metadata: page-specific title and description, canonical URL, Open Graph / Twitter card tags and JSON-LD structured data. Pages can override any of it from front matter: description: "..." meta description, og:description, twitter:description images: ["..."] social card image (absolute or site-relative URL) */ -}} {{- $site := .Site -}} {{- $author := $site.Params.author | default $site.Title -}} {{- /* Title: " · " everywhere except the homepage. */ -}} {{- $title := printf "%s · %s" .Title $site.Title -}} {{- if .IsHome -}} {{- $title = printf "%s · %s" $site.Title $site.Params.description -}} {{- end -}} {{- /* Description: front matter first, else the page's own opening prose. */ -}} {{- $description := "" -}} {{- if .Description -}} {{- $description = .Description -}} {{- else if .IsPage -}} {{- /* Drop the table-of-contents aside so the summary starts at real prose. */ -}} {{- $description = .Content | replaceRE `(?s)
.*?
` "" -}} {{- else if .Data.Terms -}} {{- $description = printf "Every topic written about on %s — %s" $site.Title $site.Params.description -}} {{- else if .Data.Singular -}} {{- $description = printf "Every post tagged “%s” on %s — %s" .Title $site.Title $site.Params.description -}} {{- end -}} {{- $description = $description | plainify | htmlUnescape | replaceRE `\s+` " " | replaceRE `^\s+|\s+$` "" | truncate 160 -}} {{- /* Pages with no prose of their own (list pages, link digests) fall back to the site blurb. */ -}} {{- if eq $description "" -}} {{- $description = $site.Params.description -}} {{- end -}} {{- /* Social card image: front matter first, else the site-wide card (see extra/og-card). */ -}} {{- $image := "og-card.jpg" | absURL -}} {{- with .Params.images -}} {{- $image = index . 0 | absURL -}} {{- end -}} {{- /* Profile links, reused as schema.org sameAs. */ -}} {{- $sameAs := slice -}} {{- range $site.Menus.main -}} {{- $sameAs = $sameAs | append .URL -}} {{- end -}} {{ $title }} {{- /* /posts/, /digests/ and /fieldguides/ all render the same archive listing as the homepage. Keep the URLs working but stop them competing with it in search results. */ -}} {{- if eq .Kind "section" }} {{- end }} {{- if .IsPage }} {{- with .Date }} {{- end }} {{- with .Lastmod }} {{- end }} {{- range .Params.tags }} {{- end }} {{- else }} {{- end }} {{- with $site.Params.twitter }} {{- end }} {{- /* JSON-LD: WebSite + Person on the homepage, BlogPosting on articles. Each branch builds its object in one single-line dict call: the CI toolchain (hugo 0.55) predates the dict-merging function, and its template lexer rejects newlines inside an action. */ -}} {{- $person := dict "@type" "Person" "name" $author "url" $site.BaseURL "sameAs" $sameAs -}} {{- $webSite := dict "@type" "WebSite" "name" $site.Title "url" $site.BaseURL -}} {{- $ld := dict "@context" "https://schema.org" -}} {{- if .IsHome -}} {{- $ld = dict "@context" "https://schema.org" "@type" "WebSite" "name" $site.Title "url" $site.BaseURL "description" $description "author" $person "publisher" $person -}} {{- else if and .IsPage (ne .Type "digests") -}} {{- $ld = dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" .Title "description" $description "url" .Permalink "image" $image "mainEntityOfPage" (dict "@type" "WebPage" "@id" .Permalink) "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "wordCount" .WordCount "keywords" (delimit (.Params.tags | default slice) ", ") "author" $person "publisher" $person -}} {{- else -}} {{- $ld = dict "@context" "https://schema.org" "@type" "CollectionPage" "name" .Title "description" $description "url" .Permalink "isPartOf" $webSite -}} {{- end -}}
